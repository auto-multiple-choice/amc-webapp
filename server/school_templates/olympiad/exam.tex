\centerline{\bfseries\Large Notation}
\begin{itemize}
  \item For a set $A$, $x\in A$ means ``$x$ is an element of $A$''; $|A|$ is the number of elements of $A$.
  \item $A \setminus B$ is the set of elements of $A$ that are not in $B$.
  \item $\mathbb{N}=\{1,2,3,\ldots\}$ is the set of natural numbers, $\mathbb{Z}$ the set of integers and $\mathbb{R}$ the set of real numbers.
  \item For $a,b\in\mathbb{R}$: $[a,b]=\{x\in\mathbb{R}\mid a\le x\le b\}$ and $(a,b)=\{x\in\mathbb{R}\mid a<x<b\}$; $a\mid b$ means ``$a$ divides $b$''.
  \item $n!=n\cdot(n-1)\cdots 2\cdot 1$, and $\lfloor x\rfloor$ is the largest integer not greater than $x$.
  \item $[ABC]$ is the area of triangle $ABC$.
\end{itemize}
