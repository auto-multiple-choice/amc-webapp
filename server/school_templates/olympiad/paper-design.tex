% Olympiad paper: the layout of the question paper's header, the OMR
% sheet and the rough-work pages. The values (\amcuiExamName, ...) come

% ------ Settings ------
\setlength{\parindent}{0pt}

% ------ The paper code (A, B, C...) from the copy number ------
\makeatletter

\newcommand{\makeAlph}[1]{\@Alph{#1}}
\makeatother
\newcommand{\paperCode}{\makeAlph{\slno}}

% ------ Small pieces ------
\newcount\amcuiCount
\newcommand{\bbox}[1]{\fcolorbox{black!40}{black!40}{\color{white}#1}}
\newcommand{\amcuiBox}{\framebox[6.5mm]{\rule{0pt}{7mm}}\:}
\newcommand{\amcuiDigitBoxes}[1]{\amcuiCount=0 \loop\ifnum\amcuiCount<#1 \amcuiBox\advance\amcuiCount by 1 \repeat}
\newcommand{\amcuiPrefixBoxes}[1]{\StrLen{#1}[\amcuiLen]\amcuiCount=1
  \loop\ifnum\amcuiCount<\numexpr\amcuiLen+1\relax
    \StrChar{#1}{\the\amcuiCount}[\amcuiChar]\bbox{\large\amcuiChar}\
    \advance\amcuiCount by 1 \repeat}
\newcommand{\amcuiLogo}[2]{\ifthenelse{\equal{#1}{}}{}{\IfFileExists{#1}{\includegraphics[height=#2]{#1}}{}}}
\newcommand{\amcuiPoint}[1]{\par\noindent\amcuiItem #1\par\vspace{1ex}}
\newcommand{\addcentertext}[2]{\vspace{0.1\baselineskip}%
  \centerline{\rule{#1\linewidth}{.4pt}\dbox{#2}\rule{#1\linewidth}{.4pt}}\vspace{0.25\baselineskip}}

% ------ Section headings (the Paper design panel writes them) ------
\newcommand{\amcuiSectionHeading}[1]{\par\vspace{1ex}{\centering\bfseries\Large #1\par}\vspace{2ex}\hrule\vspace*{2ex}}
\newcommand{\amcuiSectionText}[1]{\noindent #1\par\vspace*{2ex}\hrule\vspace*{2ex}}
